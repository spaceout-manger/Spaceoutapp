// ═══ WEEK GRID ═══
async function buildWeek(id) {
  const days=['SUN','MON','TUE','WED','THU','FRI','SAT'];
  const today=new Date(), sow=new Date(today); sow.setDate(today.getDate()-today.getDay());
  const eow=new Date(sow); eow.setDate(sow.getDate()+6);
  const sowStr=sow.toISOString().split('T')[0], eowStr=eow.toISOString().split('T')[0];
  const {data}=await db.from('attendance').select('*').gte('date',sowStr).lte('date',eowStr);
  const el=document.getElementById(id); if(!el)return;
  el.innerHTML='';
  let weekTotal=0;
  for(let i=0;i<7;i++){
    const d=new Date(sow); d.setDate(sow.getDate()+i);
    const ds=d.toISOString().split('T')[0];
    const recs=data?data.filter(a=>a.date===ds):[];
    const hrs=recs.reduce((s,r)=>s+(parseFloat(r.hours)||0),0);
    weekTotal+=hrs;
    const cls=hrs>=6?'worked':hrs>0?'partial':'';
    el.innerHTML+=`<div class="day-block ${cls}"><div class="day-name">${days[i]}</div><div class="day-hrs" style="color:${hrs>=6?'var(--green)':hrs>0?'var(--orange)':'var(--text4)'}">${hrs>0?hrs.toFixed(1):'—'}</div></div>`;
  }
  // Update week totals
  const tEl=document.getElementById('att-total'); if(tEl)tEl.textContent=weekTotal.toFixed(1);
  const badge=document.getElementById('att-rate-badge');
  const note=document.getElementById('att-rate-note');
  const dashBadge=document.getElementById('d-week-rate');
  const txt=weekTotal>=30?'$800 FLAT RATE':`$${(weekTotal*20).toFixed(0)} @ $20/hr`;
  const cls2=weekTotal>=30?'badge-green':'badge-orange';
  if(badge)badge.innerHTML=`<span class="badge ${cls2}" style="font-size:13px">${txt}</span>`;
  if(note)note.textContent=weekTotal>=30?'✅ Regular week':'⚠️ Reduced week';
  if(dashBadge)dashBadge.innerHTML=`<span class="badge ${cls2}" style="font-size:13px">${txt}</span>`;
  const dhEl=document.getElementById('d-hours');
  if(dhEl)dhEl.textContent=weekTotal.toFixed(1);
  const hBar=document.getElementById('d-hrs-bar');
  if(hBar){hBar.style.width=Math.min(100,(weekTotal/30)*100)+'%';hBar.style.background=weekTotal>=30?'var(--green)':'var(--accent)';}
  return weekTotal;
}

async function saveAtt(){

async function syncCloverNow(){
  const mid=localStorage.getItem('clover_mid');
  const token=localStorage.getItem('clover_token');
  const env=localStorage.getItem('clover_env')||'https://api.clover.com';
  if(!mid||!token){toast('Set Clover credentials first','warn');showPanel('clover',null);return;}
  document.getElementById('clover-status').innerHTML='<span style="color:var(--accent)">🔄 Syncing from Clover...</span>';
  setSyncStatus('syncing','Syncing Clover...');
  try{
    // Sync daily payments/sales
    const today=new Date().toISOString().split('T')[0];
    const startOfDay=new Date(); startOfDay.setHours(0,0,0,0);
    const endOfDay=new Date(); endOfDay.setHours(23,59,59,999);
    const payRes=await fetch(`${env}/v3/merchants/${mid}/payments?filter=createdTime>=${startOfDay.getTime()}&filter=createdTime<=${endOfDay.getTime()}&expand=order`,{headers:{'Authorization':`Bearer ${token}`,'Content-Type':'application/json'}});
    if(!payRes.ok)throw new Error(`Clover API error: ${payRes.status} ${payRes.statusText}`);
    const payData=await payRes.json();
    const payments=payData.elements||[];
    const grossSales=payments.filter(p=>p.result==='SUCCESS').reduce((s,p)=>s+(p.amount||0),0)/100;
    const netSales=payments.filter(p=>p.result==='SUCCESS').reduce((s,p)=>s+(p.netAmount||p.amount||0),0)/100;
    const transCount=payments.filter(p=>p.result==='SUCCESS').length;
    // Save to Supabase
    await db.from('daily_sales').upsert({date:today,gross_sales:grossSales,net_sales:netSales,transactions:transCount,source:'clover'},{onConflict:'date'});
    // Sync employee timesheets
    const empRes=await fetch(`${env}/v3/merchants/${mid}/shifts?filter=inTime>=${startOfDay.getTime()}&expand=employee`,{headers:{'Authorization':`Bearer ${token}`}});
    if(empRes.ok){
      const empData=await empRes.json();
      const shifts=empData.elements||[];
      for(const shift of shifts){
        if(!shift.inTime)continue;
        const shiftDate=new Date(shift.inTime).toISOString().split('T')[0];
        const inT=new Date(shift.inTime).toTimeString().slice(0,5);
        const outT=shift.outTime?new Date(shift.outTime).toTimeString().slice(0,5):null;
        const hours=outT?((shift.outTime-shift.inTime)/3600000):null;
        const empName=shift.employee?.name||'Employee';
        await db.from('attendance').upsert({date:shiftDate,employee_name:empName,check_in:inT,check_out:outT,hours,source:'clover'},{onConflict:'date,employee_name,check_in'}).catch(()=>{});
      }
    }
    const syncTime=new Date().toLocaleTimeString();
    localStorage.setItem('last_clover_sync',syncTime);
    document.getElementById('last-sync-time').textContent='Last sync: '+syncTime;
    document.getElementById('clover-status').innerHTML=`<span style="color:var(--green)">✅ Synced! Sales: $${grossSales.toFixed(2)} | Transactions: ${transCount}</span>`;
    setSyncStatus('','Connected');
    toast(`Clover synced! $${grossSales.toFixed(2)} in sales today`);
    loadSales();loadAtt();buildWeek('week-dash');loadDash();
  }catch(e){
    document.getElementById('clover-status').innerHTML=`<span style="color:var(--red)">❌ Error: ${e.message}. Check your API token and Merchant ID.</span>`;
    setSyncStatus('error','Clover Error');
    toast('Clover sync failed: '+e.message,'error');
  }
}

// ═══ FILES ═══
function fSel(e,type){Array.from(e.target.files).forEach(f=>procFile(f,type));}
function procFile(file,type){
  const r=new FileReader();
  r.onload=(e)=>{
    const obj={id:Date.now()+Math.random(),name:file.name,size:file.size,type:file.type,dataUrl:e.target.result};
    if(type==='exp'){expFiles.push(obj);renderFList('exp-flist',expFiles,'exp');}
    else if(type==='po'){poFiles.push(obj);renderFList('po-flist',poFiles,'po');}
    else if(type==='inv'){invFiles.push(obj);renderFList('inv-flist',invFiles,'inv');}
  };
  r.readAsDataURL(file);
}
function renderFList(cid,files,type){

  const fid=parseFloat(id);
  if(type==='exp'){expFiles=expFiles.filter(f=>f.id!==fid);renderFList('exp-flist',expFiles,'exp');}
  else if(type==='po'){poFiles=poFiles.filter(f=>f.id!==fid);renderFList('po-flist',poFiles,'po');}

}

      return;
    }
    // Set default dates
    const today=new Date().toISOString().split('T')[0];
    ['att-date','exp-date','po-date','po-due','inv-date','inv-due','perf-date','pay-date','sale-date'].forEach(id=>{const el=document.getElementById(id);if(el)el.value=today;});
    const finM=document.getElementById('fin-month');if(finM)finM.value=new Date().toISOString().slice(0,7);
    // Load clover creds
    const mid=localStorage.getItem('clover_mid');const token=localStorage.getItem('clover_token');
    const midEl=document.getElementById('clover-mid');const tokEl=document.getElementById('clover-token');
    if(midEl&&mid)midEl.value=mid;if(tokEl&&token)tokEl.value=token;
    // Init
    initPerf();
    await loadDash();
    // Hide loading
    document.getElementById('loading').style.display='none';
    document.getElementById('app').style.display='block';

  }

async function installApp() {
  if (!deferredPrompt) return;
  deferredPrompt.prompt();
  const { outcome } = await deferredPrompt.userChoice;
  if (outcome === 'accepted') {
    toast('App installed successfully! 🎉');
    document.getElementById('install-banner')?.remove();
  }
  deferredPrompt = null;
}
